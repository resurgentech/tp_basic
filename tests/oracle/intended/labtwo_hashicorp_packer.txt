# Packages and the login user's groups are equal on both sides (admin_users =
# ["bot"], vars/labtwo_hashicorp_packer.toml and the Ansible --extra-vars), so
# only the `apt sources` section differs (membership is the `groups` section).
#
# Ansible writes a one-line legacy /etc/apt/sources.list.d/hashicorp.list with
# a hard-coded architecture; telepatch removes that file (a stock VM has none)
# and writes the deb822 hashicorp.sources, with the architecture read on the
# host and the armored key at /etc/apt/keyrings/hashicorp.asc named in
# Signed-By (the Ansible side trusts a dearmored key in trusted.gpg.d, which
# the oracle does not capture).
-apt sources|-- /etc/apt/sources.list.d/hashicorp.list
-apt sources|deb [arch=amd64] https://apt.releases.hashicorp.com noble main
+apt sources|-- /etc/apt/sources.list.d/hashicorp.sources
+apt sources|Types: deb
+apt sources|URIs: https://apt.releases.hashicorp.com
+apt sources|Suites: noble
+apt sources|Components: main
+apt sources|Architectures: amd64
+apt sources|Signed-By: /etc/apt/keyrings/hashicorp.asc
